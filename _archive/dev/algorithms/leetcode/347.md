class Solution {
    public int[] topKFrequent(int[] nums, int k) {
        // 1단계: 빈도수 카운트
        Map<Integer, Integer> count = new HashMap<>();
        for (int num : nums) {
            count.merge(num, 1, Integer::sum);
        }

        // 2단계: 크기 k의 최소 힙 유지 (빈도수 기준)
        PriorityQueue<Integer> heap = new PriorityQueue<>(
                (a, b) -> count.get(a) - count.get(b)  // 빈도 오름차순
        );

        for (int num : count.keySet()) {
            heap.offer(num);
            if (heap.size() > k) {
                heap.poll();  // 빈도가 가장 낮은 원소 제거
            }
        }

        // 3단계: 힙에 남은 k개가 정답
        int[] result = new int[k];
        for (int i = 0; i < k; i++) {
            result[i] = heap.poll();
        }
        return result;
    }
}
