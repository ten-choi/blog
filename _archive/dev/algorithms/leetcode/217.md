import java.util.HashMap;
import java.util.Map;

class Solution {
    public boolean containsDuplicate(int[] nums) {
        Set<Integer> seen = new HashSet<>();
        for (int num : nums) {
            // add() returns false if the element already exists
            if (!seen.add(num)) {
                return true;
            }
        }
        return false;
    }
}
