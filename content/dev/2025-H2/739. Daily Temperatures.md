int n = temperatures.length;
int[] answer = new int[n]; //all zeros by default
Deque<Integer> stack = new ArrayDeque<>(); // stores indices (keeps temperatures in decreasing order)

		for (int i = 0; i < n; i++) {
			// if the current temperature is higher than the stack top's -> that day's answer is settled
			while (!stack.isEmpty() && temperatures[i] > temperatures[stack.peek()]) {
				int prevIndex = stack.pop();
				answer[prevIndex] = i - prevIndex;
			}
			stack.push(i);
		}

		System.out.println(Arrays.toString(answer));
