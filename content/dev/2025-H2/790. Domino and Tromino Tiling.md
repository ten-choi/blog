fun numTilings(n: Int): Int {
    val MOD = 1_000_000_007

    // dp[i]: number of ways to tile a 2 x i board
    if (n == 0) return 1
    if (n == 1) return 1
    if (n == 2) return 2

    val dp = IntArray(n + 1)
    dp[0] = 1 // empty board
    dp[1] = 1 // one domino
    dp[2] = 2 // two dominoes or one tromino

    for (i in 3..n) {
        // recurrence: dp[i] = 2*dp[i-1] + dp[i-3]
        dp[i] = (((2L * dp[i - 1]) % MOD + dp[i - 3]) % MOD).toInt()
    }

    return dp[n]
}
