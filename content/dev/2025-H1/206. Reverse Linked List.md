class Solution {
    fun reverseList(head: ListNode?): ListNode? {
        var prev: ListNode? = null
        var curr = head

        while (curr != null) {
                     2          3
            val nextTemp = curr.next  // save the node after the current one
              null         null
            curr.next = prev          // flip the direction
              1      1
            prev = curr               // move prev
              2      2
            curr = nextTemp           // move curr
        }

        return prev  // prev becomes the new head
    }
}
