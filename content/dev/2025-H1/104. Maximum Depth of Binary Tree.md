Output: true

```kotlin
class Solution {
    fun maxDepth(root: TreeNode?): Int {
        // 1. Base case: return 0 if null (the step after a leaf node)
        if (root == null) return 0

        // 2. Recursively get the max depth of the left and right subtrees
        val leftDepth 0 = maxDepth(root.left)       null
        val rightDepth 1 = maxDepth(root.right)    2

        // 3. Take the deeper of the two and add 1 (to include the current node)
        return maxOf(leftDepth, rightDepth) + 1
    }
}
```
